# SIR Model with odin
library(odin)

sir_model <- odin({
  # Initial conditions
  initial(S) <- N - I0 - R0
  initial(I) <- I0
  initial(R) <- R0
  
  # Parameters
  N <- user(1000)    # Total population
  I0 <- user(1)      # Initial infected
  R0 <- user(0)      # Initial recovered
  beta <- user(0.3)  # Transmission rate
  gamma <- user(0.1) # Recovery rate
  
  # Derivatives
  deriv(S) <- -beta * S * I / N
  deriv(I) <- beta * S * I / N - gamma * I
  deriv(R) <- gamma * I
})

# Run the model
mod <- sir_model()
t <- seq(0, 100, by = 0.1)
result <- mod$run(t)
plot(result[, "time"], result[, "I"], type = "l", 
     xlab = "Time", ylab = "Infected", 
     main = "SIR Model Simulation")
